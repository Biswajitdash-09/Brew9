## Architecture & Final Changelog (v4)

| # | Item | Resolution in v4 |
|---|------|------------------|
| 1 | **Installed Clerk API Verification** | Will inspect `@clerk/nextjs` exports after install to confirm exact Next.js 16 proxy / middleware signatures (`src/proxy.ts`, `clerkMiddleware`, `createRouteMatcher`). |
| 2 | **Zero-Fallback Auth Config** | `CLERK_JWT_ISSUER_DOMAIN` is strictly required in `convex/auth.config.ts` without any invented fallback URLs. |
| 3 | **Separated Client & Server Env** | Split into `src/lib/env/client.ts` and `src/lib/env/server.ts` with `server-only` import guards to prevent leakage into client bundles. |
| 4 | **Transaction-Safe Rate Limiter** | Leverages Convex mutation atomicity for concurrent consistency; user-facing errors are generic ("Too many requests. Please try again in a few minutes.") with structured error codes. |
| 5 | **Session-Guarded User Sync** | `AuthSyncProvider` uses session-scoped ref tracking to ensure `ensureCurrentUser` runs **strictly once** per login session; never mutates on every render and fails gracefully without blocking the UI. |
| 6 | **Structured Weekly Hours** | Locations use a scalable `weeklySchedule` array schema (Option A) with `dayOfWeek`, `openTime`, `closeTime`, `isClosed`, and `specialNotice`. |
| 7 | **Integer INR Monetary Rule** | Authoritative prices stored strictly as integer rupees (`₹30`, `₹40`, `₹50`). Explicit rule: if fractional paisa is needed in future, monetary representation must be redesigned. |
| 8 | **Runtime CSP Verification** | CSP will be validated live in preview against Clerk, Convex WebSockets, Cloudflare Turnstile, and self-hosted fonts before final deployment. |
| 9 | **Testing & Quality Gates** | Added explicit `npx tsc --noEmit`, `npm audit --audit-level=high`, and manual a11y tests (focus trap, drawer escape, screen-reader error announcements). |

### PHASE 5: Convex Database Schema & Backend

#### [NEW] `convex/schema.ts`

```typescript
import { defineSchema, defineTable } from "convex/server";
import { v } from "convex/values";

export default defineSchema({
  users: defineTable({
    clerkId: v.string(),
    email: v.string(),
    name: v.string(),
    imageUrl: v.optional(v.string()),
    role: v.union(v.literal("user"), v.literal("admin")),
    createdAt: v.number(),
    updatedAt: v.number(),
  })
    .index("by_clerkId", ["clerkId"])
    .index("by_email", ["email"])
    .index("by_role", ["role"]),

  categories: defineTable({
    name: v.string(),
    slug: v.string(),
    displayOrder: v.number(),
  })
    .index("by_slug", ["slug"])
    .index("by_displayOrder", ["displayOrder"]),

  menuItems: defineTable({
    name: v.string(),
    description: v.string(),
    price: v.number(), // Authoritative integer in whole rupees (₹)
    categorySlug: v.string(),
    imageUrl: v.optional(v.string()),
    isAvailable: v.boolean(),
    isFeatured: v.boolean(),
    displayOrder: v.number(),
    createdAt: v.number(),
    updatedAt: v.number(),
  })
    .index("by_categorySlug", ["categorySlug"])
    .index("by_featured", ["isFeatured"])
    .index("by_displayOrder", ["displayOrder"]),

  locations: defineTable({
    name: v.string(),
    slug: v.string(),
    address: v.optional(v.string()),       // Optional until real kiosk is commissioned
    city: v.string(),
    state: v.string(),
    postalCode: v.optional(v.string()),
    latitude: v.optional(v.number()),      // Optional: No fake coordinates
    longitude: v.optional(v.number()),
    phone: v.optional(v.string()),
    status: v.union(v.literal("open"), v.literal("coming_soon"), v.literal("temporarily_closed")),
    imageUrl: v.optional(v.string()),
    // Scalable structured weekly schedule (Option A)
    weeklySchedule: v.optional(
      v.array(
        v.object({
          dayOfWeek: v.number(), // 0 = Sunday, 1 = Monday ... 6 = Saturday
          openTime: v.string(),  // e.g. "09:00"
          closeTime: v.string(), // e.g. "21:00"
          isClosed: v.boolean(),
        })
      )
    ),
    specialNotice: v.optional(v.string()),
    createdAt: v.number(),
    updatedAt: v.number(),
  })
    .index("by_slug", ["slug"])
    .index("by_status", ["status"]),

  favorites: defineTable({
    userId: v.id("users"),
    menuItemId: v.id("menuItems"),
    createdAt: v.number(),
  })
    .index("by_userId", ["userId"])
    .index("by_menuItemId", ["menuItemId"])
    .index("by_userId_menuItemId", ["userId", "menuItemId"]),

  feedback: defineTable({
    userId: v.id("users"),
    rating: v.number(),
    message: v.string(),
    locationId: v.optional(v.id("locations")),
    createdAt: v.number(),
  })
    .index("by_userId", ["userId"])
    .index("by_createdAt", ["createdAt"]),

  rateLimits: defineTable({
    identifier: v.string(),
    action: v.string(),
    windowStart: v.number(),
    count: v.number(),
  }).index("by_identifier_action", ["identifier", "action"]),
});
```

#### [NEW] `src/lib/providers/auth-sync-provider.tsx` (Session-Guarded, Single-Execution)

```typescript
"use client";

import { useEffect, useRef } from "react";
import { useUser } from "@clerk/nextjs";
import { useMutation } from "convex/react";
import { api } from "@/convex/_generated/api";

export function AuthSyncProvider({ children }: { children: React.ReactNode }) {
  const { isSignedIn, user, isLoaded } = useUser();
  const ensureUser = useMutation(api.users.ensureCurrentUser);
  const syncedUserIdRef = useRef<string | null>(null);

  useEffect(() => {
    if (!isLoaded || !isSignedIn || !user) return;
    
    // Execute strictly once per user login session
    if (syncedUserIdRef.current === user.id) return;
    syncedUserIdRef.current = user.id;

    ensureUser().catch((err) => {
      console.error("Non-blocking auth sync warning:", err);
      // Reset ref so it can retry on subsequent user action if needed
      syncedUserIdRef.current = null;
    });
  }, [isLoaded, isSignedIn, user, ensureUser]);

  return <>{children}</>;
}
```

### PHASE 11: Feedback System & Transactionally Safe Rate Limiting

#### [NEW] `convex/lib/rateLimit.ts`

```typescript
import { MutationCtx } from "../_generated/server";

interface RateLimitRule {
  windowMs: number;
  maxRequests: number;
}

const RULES: Record<string, RateLimitRule> = {
  "feedback:submit": { windowMs: 60 * 60 * 1000, maxRequests: 3 }, // 3/hour
  "favorites:toggle": { windowMs: 60 * 1000, maxRequests: 30 },     // 30/min
};

/**
 * Transaction-safe sliding window rate limit enforcement inside a Convex mutation.
 */
export async function enforceRateLimit(
  ctx: MutationCtx,
  identifier: string,
  action: string
): Promise<void> {
  const rule = RULES[action];
  if (!rule) return;

  const now = Date.now();
  const windowBoundary = now - rule.windowMs;

  const record = await ctx.db
    .query("rateLimits")
    .withIndex("by_identifier_action", (q) =>
      q.eq("identifier", identifier).eq("action", action)
    )
    .unique();

  if (!record || record.windowStart < windowBoundary) {
    if (record) {
      await ctx.db.patch(record._id, { windowStart: now, count: 1 });
    } else {
      await ctx.db.insert("rateLimits", { identifier, action, windowStart: now, count: 1 });
    }
    return;
  }

  if (record.count >= rule.maxRequests) {
    // Generic, non-leaking user error message
    throw new Error("Too many requests. Please try again in a few minutes.");
  }

  await ctx.db.patch(record._id, { count: record.count + 1 });
}
```

### PHASE 14: Security Hardening & Runtime-Verified CSP

#### `next.config.ts`

```typescript
import type { NextConfig } from "next";

const isProduction = process.env.NODE_ENV === "production";

// Minimal, hardened CSP (to be validated in runtime/preview)
const cspHeader = `
  default-src 'self';
  script-src 'self' 'unsafe-inline' https://*.clerk.accounts.dev https://challenges.cloudflare.com;
  connect-src 'self' https://*.clerk.accounts.dev https://*.convex.cloud wss://*.convex.cloud;
  img-src 'self' data: blob: https://img.clerk.com https://*.clerk.accounts.dev;
  frame-src https://*.clerk.accounts.dev https://challenges.cloudflare.com;
  frame-ancestors 'none';
  worker-src 'self' blob:;
  base-uri 'self';
  form-action 'self';
`.replace(/\s{2,}/g, " ").trim();

const securityHeaders = [
  { key: "Content-Security-Policy", value: cspHeader },
  { key: "X-Content-Type-Options", value: "nosniff" },
  { key: "Referrer-Policy", value: "strict-origin-when-cross-origin" },
  { key: "Permissions-Policy", value: "camera=(), microphone=(), geolocation=()" },
  ...(isProduction
    ? [{ key: "Strict-Transport-Security", value: "max-age=63072000; includeSubDomains; preload" }]
    : []),
];

const nextConfig: NextConfig = {
  async headers() {
    return [{ source: "/(.*)", headers: securityHeaders }];
  },
};

export default nextConfig;
```
